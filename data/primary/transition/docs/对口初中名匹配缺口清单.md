## 核验方式

```bash
node - <<'NODE'
const fs = require('fs');
const xs = JSON.parse(fs.readFileSync('data/primary/transition/dist/xiaoshengchu_2026.json'));
const counts = new Map();
for (const r of xs.records) for (const name of r.feed_unresolved || []) {
  counts.set(name, (counts.get(name) || 0) + 1);
}
console.table([...counts].map(([name, records]) => ({ name, records })));
NODE
```
